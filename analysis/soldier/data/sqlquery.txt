WITH class_rounds AS (
  SELECT
    player_key AS steam_id,
    match_id,
    round_num,
    lower(class_name) AS class_name,
    seconds
  FROM match_player_classes
),
player_summary AS (
  SELECT
    steam_id,
    COUNT(DISTINCT match_id) AS total_matches,
    COUNT(*) AS total_class_rounds,
    SUM(seconds) AS total_seconds,
    SUM(CASE WHEN class_name = 'scout' THEN seconds ELSE 0 END) AS scout_seconds,
    SUM(CASE WHEN class_name = 'sniper' THEN seconds ELSE 0 END) AS sniper_seconds,
    SUM(CASE WHEN class_name = 'soldier' THEN seconds ELSE 0 END) AS soldier_seconds,
    SUM(CASE WHEN class_name = 'demoman' THEN seconds ELSE 0 END) AS demoman_seconds,
    SUM(CASE WHEN class_name = 'medic' THEN seconds ELSE 0 END) AS medic_seconds,
    SUM(CASE WHEN class_name = 'hwguy' THEN seconds ELSE 0 END) AS hwguy_seconds,
    SUM(CASE WHEN class_name = 'pyro' THEN seconds ELSE 0 END) AS pyro_seconds,
    SUM(CASE WHEN class_name = 'spy' THEN seconds ELSE 0 END) AS spy_seconds,
    SUM(CASE WHEN class_name = 'engineer' THEN seconds ELSE 0 END) AS engineer_seconds
  FROM class_rounds
  GROUP BY steam_id
),
player_stats AS (
  SELECT
    cr.steam_id,
    SUM(COALESCE(mprs.suicides, 0)) AS suicides,
    SUM(COALESCE(mprs.team_damage, 0)) AS team_damage,
    SUM(COALESCE(mprs.team_kills, 0)) AS team_kills,
    SUM(COALESCE(mprs.enemy_damage, 0)) AS enemy_damage,
    SUM(COALESCE(mprs.self_damage, 0)) AS self_damage,
    SUM(COALESCE(mprs.kills, 0)) AS kills,
    SUM(COALESCE(mprs.deaths_by_enemy, 0)) AS deaths_by_enemy,
    SUM(COALESCE(mprs.deaths_by_team, 0)) AS deaths_by_team,
    SUM(COALESCE(mprs.flag_captures, 0)) AS flag_captures,
    SUM(COALESCE(mprs.flag_touches, 0)) AS flag_touches,
    SUM(COALESCE(mprs.conc_jumps, 0)) AS conc_jumps
  FROM class_rounds cr
  LEFT JOIN match_player_round_stats mprs
    ON mprs.player_key = cr.steam_id
   AND mprs.match_id = cr.match_id
   AND mprs.round_num = cr.round_num
  GROUP BY cr.steam_id
),
names AS (
  SELECT
    player_key AS steam_id,
    MAX(display_name) AS game_name
  FROM match_player_stats
  GROUP BY player_key
),
discord_links AS (
  SELECT
    steam_id,
    MAX(discord_id) AS discord_id,
    MAX(display_name) AS discord_name
  FROM player_steam_ids
  GROUP BY steam_id
),
dates AS (
  SELECT
    cr.steam_id,
    MIN(m.created_at) AS first_match_ts,
    MAX(m.created_at) AS last_match_ts,
    datetime(MIN(m.created_at), 'unixepoch') AS first_match_date,
    datetime(MAX(m.created_at), 'unixepoch') AS last_match_date
  FROM class_rounds cr
  JOIN matches m ON m.match_id = cr.match_id
  GROUP BY cr.steam_id
)
SELECT
  ps.steam_id,
  dl.discord_id,
  dl.discord_name,
  COALESCE(dl.discord_name, n.game_name, ps.steam_id) AS display_name,
  COALESCE(n.game_name, ps.steam_id) AS game_name,

  ps.total_matches,
  ps.total_class_rounds,
  ps.total_seconds,
  ROUND(ps.total_seconds / 60.0, 2) AS total_minutes,
  ROUND(ps.total_seconds / 3600.0, 2) AS total_hours,

  ROUND(ps.scout_seconds / 3600.0, 2) AS scout_hours,
  ROUND(ps.sniper_seconds / 3600.0, 2) AS sniper_hours,
  ROUND(ps.soldier_seconds / 3600.0, 2) AS soldier_hours,
  ROUND(ps.demoman_seconds / 3600.0, 2) AS demoman_hours,
  ROUND(ps.medic_seconds / 3600.0, 2) AS medic_hours,
  ROUND(ps.hwguy_seconds / 3600.0, 2) AS hwguy_hours,
  ROUND(ps.pyro_seconds / 3600.0, 2) AS pyro_hours,
  ROUND(ps.spy_seconds / 3600.0, 2) AS spy_hours,
  ROUND(ps.engineer_seconds / 3600.0, 2) AS engineer_hours,

  COALESCE(st.kills, 0) AS kills,
  COALESCE(st.deaths_by_enemy, 0) AS deaths_by_enemy,
  COALESCE(st.deaths_by_team, 0) AS deaths_by_team,
  COALESCE(st.suicides, 0) AS suicides,
  COALESCE(st.enemy_damage, 0) AS enemy_damage,
  COALESCE(st.team_damage, 0) AS team_damage,
  COALESCE(st.self_damage, 0) AS self_damage,
  COALESCE(st.team_kills, 0) AS team_kills,
  COALESCE(st.flag_captures, 0) AS flag_captures,
  COALESCE(st.flag_touches, 0) AS flag_touches,
  COALESCE(st.conc_jumps, 0) AS conc_jumps,

  ROUND(COALESCE(st.suicides, 0) * 1.0 / NULLIF(ps.total_matches, 0), 3) AS suicides_per_match,
  ROUND(COALESCE(st.team_damage, 0) * 1.0 / NULLIF(ps.total_matches, 0), 2) AS team_damage_per_match,
  ROUND(COALESCE(st.team_kills, 0) * 1.0 / NULLIF(ps.total_matches, 0), 3) AS team_kills_per_match,

  d.first_match_ts,
  d.last_match_ts,
  d.first_match_date,
  d.last_match_date

FROM player_summary ps
LEFT JOIN player_stats st ON st.steam_id = ps.steam_id
LEFT JOIN names n ON n.steam_id = ps.steam_id
LEFT JOIN discord_links dl ON dl.steam_id = ps.steam_id
LEFT JOIN dates d ON d.steam_id = ps.steam_id
ORDER BY ps.total_matches DESC;
